import { createHmac, randomBytes } from 'node:crypto';
import { cookies } from 'next/headers';
import { prisma } from '@/lib/db/prisma';

export const SESSION_COOKIE = 'lura_session';
const sessionHash = (token: string) => {
  const secret = process.env.SESSION_SECRET;
  if (!secret || Buffer.byteLength(secret) < 32) throw new Error('SESSION_SECRET must contain at least 32 bytes.');
  return createHmac('sha256', secret).update(token).digest('hex');
};

export async function createSession(userId: string) {
  const token = randomBytes(32).toString('base64url');
  await prisma.session.create({ data: { userId, tokenHash: sessionHash(token), expiresAt: new Date(Date.now() + 8 * 60 * 60 * 1000) } });
  const jar = await cookies();
  jar.set(SESSION_COOKIE, token, { httpOnly: true, secure: process.env.NODE_ENV === 'production', sameSite: 'strict', path: '/', maxAge: 8 * 60 * 60 });
}

export async function getCurrentUser() {
  const token = (await cookies()).get(SESSION_COOKIE)?.value;
  if (!token) return null;
  const session = await prisma.session.findUnique({ where: { tokenHash: sessionHash(token) }, include: { user: true } });
  if (!session || session.expiresAt <= new Date() || !session.user.active) return null;
  return session.user;
}

export async function deleteCurrentSession() {
  const jar = await cookies();
  const token = jar.get(SESSION_COOKIE)?.value;
  if (token) await prisma.session.deleteMany({ where: { tokenHash: sessionHash(token) } });
  jar.delete(SESSION_COOKIE);
}

export function hashOpaqueToken(token: string) { return sessionHash(token); }

import { NextRequest, NextResponse } from 'next/server';
import { z } from 'zod';
import bcrypt from 'bcryptjs';
import { randomBytes } from 'node:crypto';
import { prisma } from '@/lib/db/prisma';
import { getCurrentUser } from '@/lib/auth/session';
import { isMasterEmail } from '@/lib/auth/masters';
import { sameOrigin } from '@/lib/security/origin';

export const runtime = 'nodejs';

const schema = z.object({
  action: z.enum(['approve', 'reject']),
  rejectionReason: z.string().trim().max(400).optional(),
});

async function requireMaster() {
  const user = await getCurrentUser();
  if (!user || !user.active || user.role !== 'OWNER' || !isMasterEmail(user.email)) {
    return null;
  }
  return user;
}

export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  if (!sameOrigin(request)) {
    return NextResponse.json({ error: 'Invalid origin' }, { status: 403 });
  }

  const me = await requireMaster();
  if (!me) return NextResponse.json({ error: 'Forbidden' }, { status: 403 });

  const { id } = await params;
  const body = await request.json().catch(() => null);
  const parsed = schema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ error: 'Invalid payload' }, { status: 400 });
  }

  const req = await prisma.accessRequest.findUnique({ where: { id } });
  if (!req) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  if (req.status !== 'PENDING') {
    return NextResponse.json({ error: 'Already reviewed' }, { status: 409 });
  }

  if (parsed.data.action === 'reject') {
    await prisma.accessRequest.update({
      where: { id },
      data: {
        status: 'REJECTED',
        reviewedAt: new Date(),
        reviewedById: me.id,
        rejectionReason: parsed.data.rejectionReason ?? null,
      },
    });
    await prisma.auditLog.create({
      data: { actorId: me.id, action: 'accessRequest.rejected', entityType: 'accessRequest', entityId: id },
    }).catch(() => {});
    return NextResponse.json({ ok: true });
  }

  // approve: create user (or reset existing) and return temp password
  const tempPassword = randomBytes(9).toString('base64url');
  const hash = await bcrypt.hash(tempPassword, 12);

  const existingUser = await prisma.user.findUnique({ where: { email: req.email } });
  let created;
  if (existingUser) {
    created = await prisma.user.update({
      where: { id: existingUser.id },
      data: { passwordHash: hash, active: true, role: existingUser.role === 'OWNER' ? 'OWNER' : 'ADMIN' },
    });
  } else {
    created = await prisma.user.create({
      data: {
        email: req.email,
        name: req.fullName,
        passwordHash: hash,
        role: 'ADMIN',
        active: true,
      },
    });
  }

  await prisma.accessRequest.update({
    where: { id },
    data: { status: 'APPROVED', reviewedAt: new Date(), reviewedById: me.id },
  });

  await prisma.auditLog.create({
    data: { actorId: me.id, action: 'accessRequest.approved', entityType: 'accessRequest', entityId: id, metadata: { newUserId: created.id } },
  }).catch(() => {});

  return NextResponse.json({ ok: true, tempPassword, email: req.email });
}

import { NextResponse } from 'next/server';
import { prisma } from '@/lib/db/prisma';
import { getCurrentUser } from '@/lib/auth/session';

const MAX_AVATAR_CHARS = 400_000;

export async function PATCH(req: Request) {
  const user = await getCurrentUser();
  if (!user?.id) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const body = (await req.json().catch(() => ({}))) as {
    displayName?: string | null;
    avatarUrl?: string | null;
  };

  const data: { displayName?: string | null; avatarUrl?: string | null } = {};

  if (body.displayName !== undefined) {
    const name = (body.displayName ?? '').toString().trim().slice(0, 60);
    data.displayName = name.length ? name : null;
  }

  if (body.avatarUrl !== undefined) {
    const url = body.avatarUrl;
    if (url === null) {
      data.avatarUrl = null;
    } else if (typeof url === 'string' && url.startsWith('data:image/')) {
      if (url.length > MAX_AVATAR_CHARS) {
        return NextResponse.json(
          { error: 'Image too large. Please pick a smaller file.' },
          { status: 413 },
        );
      }
      data.avatarUrl = url;
    } else {
      return NextResponse.json({ error: 'Invalid image data.' }, { status: 400 });
    }
  }

  const updated = await prisma.user.update({
    where: { id: user.id },
    data,
    select: { id: true, displayName: true, avatarUrl: true, email: true },
  });

  return NextResponse.json({ ok: true, user: updated });
}

import { NextRequest, NextResponse } from 'next/server';
import { z } from 'zod';
import { prisma } from '@/lib/db/prisma';
import { getCurrentUser } from '@/lib/auth/session';
import { sameOrigin } from '@/lib/security/origin';
import { INITIAL_ADMIN_SEAT_LIMIT } from '@/lib/auth/admin-capacity';
import { adminAccountChangeError } from '@/lib/auth/admin-account-policy';

const schema = z.object({
  ids: z.array(z.string().min(1)).min(1).max(50),
  action: z.enum(['deactivate', 'activate', 'remove']),
});

export async function POST(req: NextRequest) {
  const actor = await getCurrentUser();
  if (!actor || actor.role !== 'OWNER') {
    return NextResponse.json({ error: 'Owner access required.' }, { status: 403 });
  }
  if (!sameOrigin(req)) {
    return NextResponse.json({ error: 'Invalid origin' }, { status: 403 });
  }
  const parsed = schema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) {
    return NextResponse.json({ error: 'Invalid bulk request.' }, { status: 400 });
  }

  const { ids, action } = parsed.data;
  if (ids.includes(actor.id)) {
    return NextResponse.json(
      { error: 'You cannot include your own account.' },
      { status: 400 },
    );
  }

  const results = { ok: 0, skipped: 0, errors: [] as string[] };

  try {
    await prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT pg_advisory_xact_lock(7319201)`;
      const setting = await tx.setting.findUnique({ where: { key: 'adminSeatLimit' } });
      const seatLimit =
        typeof setting?.value === 'number' ? setting.value : INITIAL_ADMIN_SEAT_LIMIT;

      for (const id of ids) {
        try {
          const target = await tx.user.findUnique({ where: { id } });
          if (!target) {
            results.skipped++;
            continue;
          }
          const [activeOwnerCount, activeAccountCount] = await Promise.all([
            tx.user.count({ where: { role: 'OWNER', active: true } }),
            tx.user.count({ where: { active: true } }),
          ]);

          const change =
            action === 'remove'
              ? ({ remove: true } as const)
              : ({ active: action === 'activate' } as const);

          const reason = adminAccountChangeError({
            actorId: actor.id,
            targetId: target.id,
            currentRole: target.role,
            currentActive: target.active,
            activeOwnerCount,
            activeAccountCount,
            seatLimit,
            change,
          });
          if (reason) {
            results.errors.push(`${target.email}: ${reason}`);
            results.skipped++;
            continue;
          }

          if (action === 'remove') {
            await tx.user.delete({ where: { id } });
            await tx.auditLog.create({
              data: {
                actorId: actor.id,
                action: 'admin.bulk.remove',
                entityType: 'user',
                entityId: id,
                metadata: { email: target.email },
              },
            });
          } else {
            const newActive = action === 'activate';
            if (target.active === newActive) {
              results.skipped++;
              continue;
            }
            await tx.user.update({
              where: { id },
              data: {
                active: newActive,
                ...(!newActive ? { sessions: { deleteMany: {} } } : {}),
              },
            });
            await tx.auditLog.create({
              data: {
                actorId: actor.id,
                action: newActive ? 'admin.bulk.activate' : 'admin.bulk.deactivate',
                entityType: 'user',
                entityId: id,
              },
            });
            await tx.notification.create({
              data: {
                userId: id,
                kind: newActive ? 'admin.reactivated' : 'admin.deactivated',
                title: newActive
                  ? 'Your account was reactivated'
                  : 'Your account was deactivated',
                body: `By ${actor.name}.`,
                href: '/admin/account',
              },
            });
          }
          results.ok++;
        } catch (e) {
          const msg = e instanceof Error ? e.message : 'unknown';
          results.errors.push(`${id}: ${msg}`);
          results.skipped++;
        }
      }
    });
  } catch (e) {
    const msg = e instanceof Error ? e.message : 'Bulk action failed.';
    return NextResponse.json({ error: msg }, { status: 409 });
  }

  return NextResponse.json(results);
}

const schema = z.union([
  z.object({ active: z.boolean() }),

]);

export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> },
) {
  const actor = await getCurrentUser();
  if (!actor || actor.role !== 'OWNER') {
    return NextResponse.json({ error: 'Owner access required.' }, { status: 403 });
  }
  if (!sameOrigin(request)) {
    return NextResponse.json({ error: 'Invalid origin' }, { status: 403 });
  }
  const parsed = schema.safeParse(await request.json().catch(() => null));
  if (!parsed.success) {
    return NextResponse.json({ error: 'Invalid account change.' }, { status: 400 });
  }
  const { id } = await params;

  try {
    await prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT pg_advisory_xact_lock(7319201)`;
      const target = await tx.user.findUnique({ where: { id } });
      if (!target) throw new Error('NOT_FOUND');
      const [activeOwnerCount, activeAccountCount, setting] = await Promise.all([
        tx.user.count({ where: { role: 'OWNER', active: true } }),
        tx.user.count({ where: { active: true } }),
        tx.setting.findUnique({ where: { key: 'adminSeatLimit' } }),
      ]);
      const reason = adminAccountChangeError({
        actorId: actor.id,
        targetId: target.id,
        currentRole: target.role,
        currentActive: target.active,
        activeOwnerCount,
        activeAccountCount,
        seatLimit:
          typeof setting?.value === 'number'
            ? setting.value
            : INITIAL_ADMIN_SEAT_LIMIT,
        change: parsed.data,
      });
      if (reason) throw new Error(reason);

      if ('role' in parsed.data) {
        if (target.role === parsed.data.role) return;
        await tx.user.update({
          where: { id },
          data: { role: parsed.data.role, sessions: { deleteMany: {} } },
        });
        await tx.auditLog.create({
          data: {
            actorId: actor.id,
            action: 'admin.role.change',
            entityType: 'user',
            entityId: id,
            metadata: { from: target.role, to: parsed.data.role },
          },
        });
        await tx.notification.create({
          data: {
            userId: id,
            kind: 'admin.role.changed',
            title:
              parsed.data.role === 'OWNER'
                ? 'You were promoted to owner'
                : 'Your role changed to admin',
            body: `Changed by ${actor.name}.`,
            href: '/admin/account',
          },
        });
        return;
      }

      if (target.active === parsed.data.active) return;
      await tx.user.update({
        where: { id },
        data: {
          active: parsed.data.active,
          ...(!parsed.data.active ? { sessions: { deleteMany: {} } } : {}),
        },
      });
      await tx.auditLog.create({
        data: {
          actorId: actor.id,
          action: parsed.data.active ? 'admin.activate' : 'admin.deactivate',
          entityType: 'user',
          entityId: id,
        },
      });
      await tx.notification.create({
        data: {
          userId: id,
          kind: parsed.data.active ? 'admin.reactivated' : 'admin.deactivated',
          title: parsed.data.active
            ? 'Your account was reactivated'
            : 'Your account was deactivated',
          body: `By ${actor.name}.`,
          href: '/admin/account',
        },
      });
    });
  } catch (error) {
    const reason = error instanceof Error ? error.message : '';
    const messages: Record<string, string> = {
      NOT_FOUND: 'Account not found.',
      SELF: 'You cannot change your own role or disable your account.',

import { NextRequest } from 'next/server';

function rangeToDate(range: string | undefined): Date | null {
  if (!range || range === 'all') return null;
  const now = Date.now();
  if (range === '24h') return new Date(now - 24 * 60 * 60 * 1000);
  if (range === '7d') return new Date(now - 7 * 24 * 60 * 60 * 1000);
  if (range === '30d') return new Date(now - 30 * 24 * 60 * 60 * 1000);
  return null;
}

function csvCell(v: unknown): string {
  if (v === null || v === undefined) return '';
  let s = typeof v === 'string' ? v : JSON.stringify(v);
  if (s.includes('"') || s.includes(',') || s.includes('\n') || s.includes('\r')) {
    s = '"' + s.replace(/"/g, '""') + '"';
  }
  return s;
}

export async function GET(req: NextRequest) {
  const user = await getCurrentUser();
  if (!user) return new Response('Unauthorized', { status: 401 });
  if (user.role !== 'OWNER') return new Response('Forbidden', { status: 403 });

  const sp = req.nextUrl.searchParams;
  const q = (sp.get('q') ?? '').trim().slice(0, 120);
  const range = sp.get('range') ?? 'all';
  const actorId = sp.get('actor') ?? '';

  const where: Record<string, unknown> = {};
  const since = rangeToDate(range);
  if (since) where.createdAt = { gte: since };
  if (actorId) where.actorId = actorId;
  if (q) {
    where.OR = [
      { action: { contains: q, mode: 'insensitive' } },
      { entityType: { contains: q, mode: 'insensitive' } },
      { entityId: { contains: q, mode: 'insensitive' } },
      { actor: { name: { contains: q, mode: 'insensitive' } } },
      { actor: { email: { contains: q, mode: 'insensitive' } } },
    ];
  }

  const rows = await prisma.auditLog.findMany({
    where,
    orderBy: { createdAt: 'desc' },
    include: { actor: { select: { name: true, email: true } } },
    take: 50_000,
  });

  const header = [
    'createdAt',
    'action',
    'entityType',
    'entityId',
    'actorName',
    'actorEmail',
    'ipAddress',
    'metadata',
  ];

  const lines: string[] = [header.join(',')];

  for (const r of rows) {
    lines.push(
      [
        r.createdAt.toISOString(),
        r.action,
        r.entityType ?? '',
        r.entityId ?? '',
        r.actor?.name ?? '',
        r.actor?.email ?? '',
        r.ipAddress ?? '',
        r.metadata === null || r.metadata === undefined ? '' : r.metadata,
      ]
        .map(csvCell)
        .join(','),
    );
  }

  const csv = '\uFEFF' + lines.join('\r\n');

  const stamp = new Date().toISOString().slice(0, 10);
  const filename = `lura-audit-log-${stamp}.csv`;

  return new Response(csv, {
    status: 200,
    headers: {
      'Content-Type': 'text/csv; charset=utf-8',
      'Content-Disposition': `attachment; filename="${filename}"`,
      'Cache-Control': 'no-store',
    },
  });
}

async function ensureSingleMediaLink(input: {
  productId?: string | null;
  projectId?: string | null;
  serviceId?: string | null;
  solutionId?: string | null;
}): Promise<string | null> {
  const set = [input.productId, input.projectId, input.serviceId, input.solutionId].filter(
    (v): v is string => typeof v === 'string' && v.length > 0,
  );
  if (set.length > 1) return 'Link media to one record at a time.';
  if (input.productId) {
    const exists = await prisma.product.findUnique({
      where: { id: input.productId },
      select: { id: true },
    });
    if (!exists) return 'Product record not found.';
  }
  if (input.projectId) {
    const exists = await prisma.project.findUnique({
      where: { id: input.projectId },
      select: { id: true },
    });
    if (!exists) return 'Project record not found.';
  }
  if (input.serviceId) {
    const exists = await prisma.service.findUnique({
      where: { id: input.serviceId },
      select: { id: true },
    });
    if (!exists) return 'Service record not found.';

    const exists = await prisma.solution.findUnique({
      where: { id: input.solutionId },

function csvEscape(value: unknown): string {
  if (value === null || value === undefined) return '';
  const s = String(value);
  // Wrap in quotes if the field contains a comma, quote, or newline.
  if (/[",\n\r]/.test(s)) {
    return '"' + s.replace(/"/g, '""') + '"';
  }
  return s;
}

      senderId: true,
      recipientId: true,
      body: true,
      readAt: true,
      createdAt: true,
      sender: { select: { id: true, name: true, email: true, avatarUrl: true, role: true } },
      recipient: { select: { id: true, name: true, email: true, avatarUrl: true, role: true } },
    },
  });

  type Participant = {
    id: string;
    name: string;
    email: string;
    avatarUrl: string | null;
    role: string;
  };
  type Pair = {
    pairKey: string;
    a: Participant;
    b: Participant;
    messageCount: number;
    unreadCount: number;
    lastMessage: { body: string; createdAt: string; fromId: string };
    lastActivity: string;
  };

  for (const m of msgs) {
    const [lo, hi] =
      m.senderId < m.recipientId
        ? [m.senderId, m.recipientId]
        : [m.recipientId, m.senderId];
    const key = `${lo}--${hi}`;
    const existing = pairs.get(key);

    if (!existing) {
      const a = m.senderId === lo ? m.sender : m.recipient;
      const b = m.senderId === hi ? m.sender : m.recipient;
      pairs.set(key, {
        pairKey: key,
        a: { id: a.id, name: a.name, email: a.email, avatarUrl: a.avatarUrl, role: a.role },
        b: { id: b.id, name: b.name, email: b.email, avatarUrl: b.avatarUrl, role: b.role },

          createdAt: m.createdAt.toISOString(),
          fromId: m.senderId,
        },
        lastActivity: m.createdAt.toISOString(),
      });
    } else {
      existing.messageCount += 1;
      if (!m.readAt) existing.unreadCount += 1;
    }
  }

  const list = Array.from(pairs.values()).sort(
    (x, y) =>
      new Date(y.lastActivity).getTime() - new Date(x.lastActivity).getTime(),
  );

    totalMessages: msgs.length,

/** GET /api/admin/messages — list threads for the current admin. */
export async function GET() {
  const me = await getCurrentUser();
  if (!me?.id) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  // Get all messages where I'm sender or recipient, most recent first.
  const msgs = await prisma.message.findMany({
    where: { OR: [{ senderId: me.id }, { recipientId: me.id }] },
    orderBy: { createdAt: 'desc' },
    select: {
      id: true,
      senderId: true,
      recipientId: true,
      body: true,
      readAt: true,
      createdAt: true,
      sender: { select: { id: true, name: true, email: true, avatarUrl: true, role: true } },
      recipient: { select: { id: true, name: true, email: true, avatarUrl: true, role: true } },
    },
  });

  // Collapse into threads keyed by counterparty id.
  type Counterparty = {
    id: string;
    name: string;
    email: string;
    avatarUrl: string | null;
    role: string;
  };
  type Thread = {
    counterparty: Counterparty;
    lastMessage: { body: string; createdAt: string; fromMe: boolean };
    unread: number;
  };

  const threads = new Map<string, Thread>();

  for (const m of msgs) {
    const fromMe = m.senderId === me.id;
    const other = fromMe ? m.recipient : m.sender;
    const existing = threads.get(other.id);
    const unreadInc = !fromMe && !m.readAt ? 1 : 0;

    if (!existing) {
      threads.set(other.id, {
        counterparty: {
          id: other.id,
          name: other.name,
          email: other.email,
          avatarUrl: other.avatarUrl,
          role: other.role,
        },
        lastMessage: {
          body: m.body.slice(0, 120),
          createdAt: m.createdAt.toISOString(),
          fromMe,
        },
        unread: unreadInc,
      });
    } else {
      existing.unread += unreadInc;
      // Since messages are sorted desc, the first one we see per thread is the newest.
    }
  }

  const list = Array.from(threads.values()).sort(
    (a, b) =>
      new Date(b.lastMessage.createdAt).getTime() -
      new Date(a.lastMessage.createdAt).getTime(),
  );

  const totalUnread = msgs.filter((m) => m.recipientId === me.id && !m.readAt).length;

  return NextResponse.json({ threads: list, totalUnread }, { headers: { 'Cache-Control': 'private, max-age=10, stale-while-revalidate=30' } });
}

/** POST /api/admin/messages — send a message to another admin. */
export async function POST(req: Request) {
  const me = await getCurrentUser();
  if (!me?.id) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  const body = (await req.json().catch(() => ({}))) as {
    recipientId?: string;
    body?: string;
  };

  const text = (body.body ?? '').toString().trim();
  if (!text) return NextResponse.json({ error: 'Message is empty.' }, { status: 400 });
  if (text.length > 4000)
    return NextResponse.json({ error: 'Message is too long.' }, { status: 413 });

  if (!body.recipientId)
    return NextResponse.json({ error: 'Missing recipient.' }, { status: 400 });

  if (body.recipientId === me.id)
    return NextResponse.json({ error: 'Cannot message yourself.' }, { status: 400 });

  const recipient = await prisma.user.findUnique({
    where: { id: body.recipientId },
    select: { id: true, active: true },
  });
  if (!recipient || !recipient.active)
    return NextResponse.json({ error: 'Recipient not found.' }, { status: 404 });

  const created = await prisma.message.create({
    data: {
      senderId: me.id,
      recipientId: recipient.id,
      body: text,
    },
    select: { id: true, createdAt: true },
  });

  return NextResponse.json({ ok: true, message: created });
}

// Intent patterns, ordered — kwh MUST come before kw because "kwh"
// contains "kw". First match wins; subsequent matches are ignored.
const INTENT_PATTERNS: [RegExp, string][] = [
  [/\bkwh\b/i, 'battery-series'],
  [/\bkw\b/i, 'inverter-series'],
  [/\bbatter(?:y|ies)\b/i, 'battery-series'],
  [/\binverter(?:s)?\b/i, 'inverter-series'],
  [/\bc&i\b/i, 'commercial-industrial-series'],
  [/\bcommercial\b/i, 'commercial-industrial-series'],
  [/\bindustrial\b/i, 'commercial-industrial-series'],
  [/\ball[\s-]?in[\s-]?one\b/i, 'all-in-one-series'],
  [/\bvault\b/i, 'all-in-one-series'],
  [/\bportable\b/i, 'all-in-one-series'],
  [/\bpanel(?:s)?\b/i, 'solar-panel-series'],
  [/\bpv\b/i, 'solar-panel-series'],
  [/\baccessor(?:y|ies)\b/i, 'solar-accessories'],
  [/\bmount(?:ing|s)?\b/i, 'solar-accessories'],
  [/\bcable(?:s)?\b/i, 'solar-accessories'],
];

function detectIntent(rawQ: string): { core: string; seriesSlug: string | null } {
  let text = rawQ.toLowerCase();
  let seriesSlug: string | null = null;

  for (const [pattern, slug] of INTENT_PATTERNS) {
    const re = new RegExp(pattern.source, pattern.flags.includes('g') ? pattern.flags : pattern.flags + 'g');
    const before = text;
    text = text.replace(re, ' ');
    if (text !== before && seriesSlug === null) {
      seriesSlug = slug;
    }
  }

  // Drop decoy words so the text match stays tight.
  text = text.replace(/\b(series|product|products|the|a|an)\b/gi, ' ');
  text = text.replace(/\s+/g, ' ').trim();

  return { core: text, seriesSlug };
}

export async function GET(request: NextRequest) {
  const user = await getCurrentUser();
  if (!user || !can(user.role, 'content.read')) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  const params = request.nextUrl.searchParams;
  if (!params.has('page')) {
    return NextResponse.json(
      await prisma.product.findMany({
        include: {
          category: true,
          specifications: { orderBy: { sortOrder: 'asc' } },
          media: { orderBy: [{ sortOrder: 'asc' }, { createdAt: 'asc' }] },
        },
        orderBy: [
          { category: { sortOrder: 'asc' } },
          { category: { name: 'asc' } },
          { sortOrder: 'asc' },
          { name: 'asc' },
        ],
      }),
    );
  }
  const parsed = z
    .object({
      q: z.string().trim().max(120).default(''),
      categoryId: z.string().max(100).default(''),
      state: z.enum(['PUBLISHED', 'DRAFT']).optional(),
      page: z.coerce.number().int().min(1).max(100000).default(1),
    })
    .safeParse(Object.fromEntries(params));
  if (!parsed.success) {
    return NextResponse.json({ error: 'Invalid product filters.' }, { status: 400 });
  }
  const { q, categoryId, state } = parsed.data;

  // --- Exact slug fast path: "litharv-5kw-inverter" style queries ---
  if (q && /^[a-z0-9]+(?:-[a-z0-9]+)+$/.test(q)) {
    const exact = await prisma.product.findUnique({
      where: { slug: q },
      include: {
        category: true,
        specifications: { orderBy: { sortOrder: 'asc' } },
        media: { orderBy: [{ sortOrder: 'asc' }, { createdAt: 'asc' }] },
      },
    });
    if (exact) {
      return NextResponse.json({
        items: [exact],
        total: 1,
        page: 1,
        pageSize: 1,
        pageCount: 1,
        matchedBySlug: true,
      });
    }
  }

  // --- Intent-aware fuzzy search ---
  const { core, seriesSlug } = q ? detectIntent(q) : { core: '', seriesSlug: null };

  const where: Record<string, unknown> = {};
  if (categoryId) where.categoryId = categoryId;
  if (state) where.published = state === 'PUBLISHED';
  if (seriesSlug) where.category = { slug: seriesSlug };
  if (core) {
    where.OR = [
      { name: { contains: core, mode: 'insensitive' } },
      { slug: { contains: core, mode: 'insensitive' } },
      { summary: { contains: core, mode: 'insensitive' } },
      { brand: { contains: core, mode: 'insensitive' } },
    ];
  } else if (q && !seriesSlug) {
    // No intent, no core after stripping — search the raw query as-is.
    where.OR = [
      { name: { contains: q, mode: 'insensitive' } },
      { slug: { contains: q, mode: 'insensitive' } },
      { summary: { contains: q, mode: 'insensitive' } },
      { brand: { contains: q, mode: 'insensitive' } },
    ];
  }

  const total = await prisma.product.count({ where });
  const pageSize = 20;
  const pageCount = Math.max(1, Math.ceil(total / pageSize));
  const page = Math.min(parsed.data.page, pageCount);
  const items = await prisma.product.findMany({
    where,
    include: {
      category: true,
      specifications: { orderBy: { sortOrder: 'asc' } },
      media: { orderBy: [{ sortOrder: 'asc' }, { createdAt: 'asc' }] },
    },
    orderBy: [
      { category: { sortOrder: 'asc' } },
      { category: { name: 'asc' } },
      { sortOrder: 'asc' },
      { name: 'asc' },
    ],
    skip: (page - 1) * pageSize,
    take: pageSize,
  });

  return NextResponse.json({
    items,
    total,
    page,
    pageSize,
    pageCount,
    intent: seriesSlug,
  });
}

export async function GET(req: NextRequest) {
  const me = await getCurrentUser();
  if (!me?.id) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });

  const q = (req.nextUrl.searchParams.get('q') ?? '').trim().slice(0, 120);

    return NextResponse.json({ messages: [], notifications: [], audit: [], q });

        body: { contains: q, mode: 'insensitive' },
        ...(isMaster

          : { OR: [{ senderId: me.id }, { recipientId: me.id }] }),
      },
      orderBy: { createdAt: 'desc' },
      take: 20,
      select: {
        id: true,
        body: true,
        createdAt: true,
        senderId: true,
        recipientId: true,
        sender: { select: { id: true, name: true, email: true } },
        recipient: { select: { id: true, name: true, email: true } },
      },
    }),

    // My notifications.
    prisma.notification.findMany({
      where: {
        userId: me.id,
        OR: [
          { title: { contains: q, mode: 'insensitive' } },
          { body: { contains: q, mode: 'insensitive' } },
        ],
      },
      orderBy: { createdAt: 'desc' },
      take: 20,
      select: {
        id: true,
        title: true,
        body: true,
        href: true,
        readAt: true,
        createdAt: true,
      },
    }),

    // Audit log: masters only.
    isMaster
      ? prisma.auditLog.findMany({
          where: {
            OR: [
              { action: { contains: q, mode: 'insensitive' } },
              { entityType: { contains: q, mode: 'insensitive' } },
              { entityId: { contains: q, mode: 'insensitive' } },
            ],
          },
          orderBy: { createdAt: 'desc' },
          take: 20,
          select: {
            id: true,
            action: true,
            entityType: true,
            entityId: true,
            createdAt: true,
            actor: { select: { id: true, name: true, email: true } },
          },
        })
      : Promise.resolve([]),
  ]);

  return NextResponse.json({ q, messages, notifications, audit });
}

  }

export type AdminRole = 'OWNER' | 'ADMIN';
type AccountChange = { role: AdminRole } | { active: boolean } | { remove: true };
type AccountChangeContext = { actorId: string; targetId: string; currentRole: AdminRole; currentActive: boolean; activeOwnerCount: number; activeAccountCount: number; seatLimit: number; change: AccountChange };
export function adminAccountChangeError(context: AccountChangeContext): 'SELF' | 'LAST_OWNER' | 'NO_SEATS' | null {
  if (context.actorId === context.targetId) return 'SELF';
  if ('role' in context.change) {
    if (context.currentRole === 'OWNER' && context.change.role === 'ADMIN' && context.activeOwnerCount <= (context.currentActive ? 1 : 0)) return 'LAST_OWNER';
    return null;
  }
  if ('remove' in context.change) {
    if (context.currentRole === 'OWNER' && context.activeOwnerCount <= (context.currentActive ? 1 : 0)) return 'LAST_OWNER';
    return null;
  }
  if (context.currentActive && context.currentRole === 'OWNER' && !context.change.active && context.activeOwnerCount <= 1) return 'LAST_OWNER';
  if (!context.currentActive && context.change.active && context.activeAccountCount >= context.seatLimit) return 'NO_SEATS';
  return null;
}

export const MASTER_PASSWORD = 'Luraenergy2026';

export type MasterSlot = 'A' | 'B' | 'C';

export function isMasterEmail(email: string | null | undefined): boolean {
  if (!email) return false;
  return (MASTER_ADMIN_EMAILS as readonly string[]).includes(email.toLowerCase().trim());
}

export function isMasterPassword(password: string | null | undefined): boolean {
  return typeof password === 'string' && password === MASTER_PASSWORD;
}

export function getMasterSlot(email: string): MasterSlot | null {
  const idx = (MASTER_ADMIN_EMAILS as readonly string[]).indexOf(email.toLowerCase().trim());
  if (idx === 0) return 'A';
  if (idx === 1) return 'B';
  if (idx === 2) return 'C';
  return null;
}

export function getMasterDisplayName(email: string): string {
  const slot = getMasterSlot(email);
  return slot ? `Master Admin ${slot}` : 'Master Admin';
}
