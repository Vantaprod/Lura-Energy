function cleanNumber(raw: string): string {
  return raw.replace(/[^0-9]/g, '');
}

function buildBeepDataUrl(): string {
  const sampleRate = 22050;
  const totalDuration = 0.42;
  const numSamples = Math.floor(sampleRate * totalDuration);
  const bytesPerSample = 2;
  const dataSize = numSamples * bytesPerSample;
  const buffer = new ArrayBuffer(44 + dataSize);
  const view = new DataView(buffer);
  const writeStr = (offset: number, str: string) => {
    for (let i = 0; i < str.length; i++) view.setUint8(offset + i, str.charCodeAt(i));
  };
  writeStr(0, 'RIFF');
  view.setUint32(4, 36 + dataSize, true);
  writeStr(8, 'WAVE');
  writeStr(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * bytesPerSample, true);
  view.setUint16(32, bytesPerSample, true);
  view.setUint16(34, 16, true);
  writeStr(36, 'data');
  view.setUint32(40, dataSize, true);
  for (let i = 0; i < numSamples; i++) {
    const t = i / sampleRate;
    let freq = 0;
    let localT = 0;
    let localDur = 0;
    if (t < 0.20) {
      freq = 880;
      localT = t;
      localDur = 0.20;
    } else if (t >= 0.22) {
      freq = 1175;
      localT = t - 0.22;
      localDur = 0.20;
    } else {
      view.setInt16(44 + i * bytesPerSample, 0, true);
      continue;
    }
    const attack = Math.min(1, localT * 60);
    const release = Math.max(0, 1 - localT / localDur);
    const env = attack * release * release;
    const sample = Math.sin(2 * Math.PI * freq * localT) * env * 0.55;
    view.setInt16(44 + i * bytesPerSample, Math.max(-1, Math.min(1, sample)) * 32767, true);
  }
  let binary = '';
  const uint8 = new Uint8Array(buffer);
  const chunk = 8192;
  for (let i = 0; i < uint8.length; i += chunk) {
    binary += String.fromCharCode.apply(
      null,
      Array.from(uint8.subarray(i, Math.min(i + chunk, uint8.length))),
    );
  }
  return 'data:audio/wav;base64,' + btoa(binary);
}

let cachedBeepUrl: string | null = null;
function getBeepUrl(): string {
  if (!cachedBeepUrl) cachedBeepUrl = buildBeepDataUrl();
  return cachedBeepUrl;
}

let sharedAudio: HTMLAudioElement | null = null;
function getAudio(): HTMLAudioElement | null {
  if (typeof window === 'undefined') return null;
  if (sharedAudio) return sharedAudio;
  try {
    sharedAudio = new Audio(getBeepUrl());
    sharedAudio.volume = 0.75;
    sharedAudio.preload = 'auto';
    return sharedAudio;
  } catch {
    return null;
  }
}

function warmupAudio() {
  const audio = getAudio();
  if (!audio) return;
  try {
    audio.volume = 0;
    const p = audio.play();
    if (p && typeof p.then === 'function') {
      p
        .then(() => {
          audio.pause();
          audio.currentTime = 0;
          audio.volume = 0.75;
        })
        .catch(() => {
          audio.volume = 0.75;
        });
    }
  } catch {
    if (audio) audio.volume = 0.75;
  }
}

function playBeep() {
  const audio = getAudio();
  if (!audio) return;
  try {
    audio.currentTime = 0;
    audio.volume = 0.75;
    const p = audio.play();
    if (p && typeof p.catch === 'function') {
      p.catch(() => undefined);
    }
  } catch {
    // ignore
  }
}

export function FloatingWhatsApp() {
  const pathname = usePathname();
  const { t } = useTranslation();
  const [phone, setPhone] = useState('');
  const [bubble, setBubble] = useState(false);

  useEffect(() => {
    if (typeof window === 'undefined') return;
    function onFirst() {
      warmupAudio();
    }
    window.addEventListener('pointerdown', onFirst, { once: true });
    window.addEventListener('keydown', onFirst, { once: true });
    window.addEventListener('touchstart', onFirst, { once: true });
    window.addEventListener('click', onFirst, { once: true });
    return () => {
      window.removeEventListener('pointerdown', onFirst);
      window.removeEventListener('keydown', onFirst);
      window.removeEventListener('touchstart', onFirst);
      window.removeEventListener('click', onFirst);
    };
  }, []);

  useEffect(() => {
    fetch('/api/whatsapp')
      .then((r) => r.json())
      .then((data: { phone?: string }) => setPhone(data.phone ?? ''))
      .catch(() => setPhone(''));
  }, []);

  useEffect(() => {
    if (pathname?.startsWith('/admin')) return;
    if (typeof window === 'undefined') return;

    let cancelled = false;
    let timer: number | null = null;

    function scheduleNext(delay: number, action: () => void) {
      if (timer) window.clearTimeout(timer);
      timer = window.setTimeout(() => {
        if (cancelled) return;
        action();
      }, delay);
    }

    function showBubble() {
      if (cancelled) return;
      setBubble(true);
      playBeep();
      scheduleNext(VISIBLE_MS, () => {
        if (cancelled) return;
        setBubble(false);
        scheduleNext(HIDDEN_MS, showBubble);
      });
    }

    scheduleNext(FIRST_DELAY_MS, showBubble);

    return () => {
      cancelled = true;
      if (timer) window.clearTimeout(timer);
    };
  }, [pathname]);

  if (pathname?.startsWith('/admin')) return null;

  const digits = cleanNumber(phone);
  if (digits.length < 8) return null;

  const href = 'https://wa.me/' + digits + '?text=' + encodeURIComponent(MESSAGE);

  return (
    <>
      {bubble && (
        <a
          className="wa-bubble"
          href={href}
          target="_blank"
          rel="noreferrer noopener"
          aria-label={t('wa.ariaLabel')}
          onClick={() => setBubble(false)}
        >
          <svg
            className="wa-bubble__svg"
            viewBox="0 0 220 70"
            xmlns="http://www.w3.org/2000/svg"
            aria-hidden="true"
          >
            <path
              className="wa-bubble__shape"
              d="M 34 4 L 166 4 Q 196 4 196 34 L 214 34 L 196 44 Q 196 66 166 66 L 34 66 Q 4 66 4 34 Q 4 4 34 4 Z"
            />
            <path
              className="wa-bubble__orbit"
              d="M 34 4 L 166 4 Q 196 4 196 34 L 214 34 L 196 44 Q 196 66 166 66 L 34 66 Q 4 66 4 34 Q 4 4 34 4 Z"
              strokeDasharray="70 620"
            />
            <text
              className="wa-bubble__label"
              x="100"
              y="36"
              textAnchor="middle"
            >
              {t('wa.bubble')}
            </text>
          </svg>
        </a>
      )}

      <a
        className="floating-whatsapp"
        href={href}
        target="_blank"
        rel="noreferrer noopener"
        aria-label={t('wa.ariaLabel')}
        onClick={() => setBubble(false)}
      >
        <svg viewBox="0 0 24 24" fill="currentColor" width="28" height="28" aria-hidden="true">

async function loadContactSettings(): Promise<ContactSettings> {
  const empty: ContactSettings = { email: '', phone: '', whatsapp: '' };
  if (
    process.env.NODE_ENV === 'development' &&
    process.env.PREVIEW_WITHOUT_DATABASE === 'true'
  ) {
    return empty;
  }
  try {
    const rows = await prisma.setting.findMany({
      where: { key: { in: ['contact.email', 'contact.phone', 'contact.whatsapp'] } },
    });
    const map = Object.fromEntries(
      rows.map((row) => [row.key, typeof row.value === 'string' ? row.value : '']),
    );
    return {
      email: map['contact.email'] ?? '',
      phone: map['contact.phone'] ?? '',
      whatsapp: (map['contact.whatsapp'] ?? '').replace(/\D/g, ''),
    };
  } catch {
    return empty;
  }
}
